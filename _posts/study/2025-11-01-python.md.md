```c
import csv
import requests
from bs4 import BeautifulSoup      

comment_url = 'https://music.facode.cn/index.php/Home/Index/pl_list.html'
page_url = 'https://music.facode.cn/index.php/Home/Index/pl_list.html?voice_id=403778&info=2&page=1'
voice_id = 403778

# 第一步：获取总评论数和总页数
page_content = requests.get(page_url)
soup = BeautifulSoup(page_content.text, 'html.parser')

# 查找评论总数标签
com_tag = soup.find('span', 'btn__txt')  # 注意这里是btn__txt（双下划线）

if com_tag:
    # 提取评论数量（假设格式如"评论(123)"）
    com_text = com_tag.text.strip()
    try:
        # 尝试从文本中提取数字
        com_num = int(''.join(filter(str.isdigit, com_text)))
        print(f"总评论数: {com_num}")
        
        # 计算总页数（假设每页10条评论）
        page_num = com_num // 10
        if com_num % 10 != 0:
            page_num += 1
        print(f"总页数: {page_num}")
    except ValueError:
        print("无法提取评论数量，使用默认页数")
        page_num = 1  # 默认只爬取第一页
else:
    print("未找到评论数量标签，使用默认页数")
    page_num = 1  # 默认只爬取第一页

# 第二步：爬取所有页面的评论
all_comments = []

for page in range(1, page_num + 1):
    print(f"正在爬取第{page}页...")
    
    # 准备请求数据
    data = {
        'voice_id': voice_id,
        'info': 2,
        'page': page  # 这里使用当前循环的页码
    }
    
    # 发送请求
    response = requests.post(comment_url, data)
    
    # 检查请求是否成功
    if response.status_code == 200:
        try:
            # 解析JSON数据
            json_data = response.json()
            
            # 提取评论
            if 'data' in json_data:
                for comment in json_data['data']:
                    com_dic = {
                        '用户名': comment.get('user_name', '未知'),
                        '评论时间': comment.get('time', '未知'),
                        '评论内容': comment.get('content', '无内容')
                    }
                    all_comments.append(com_dic)
        except Exception as e:
            print(f"解析第{page}页数据出错: {e}")
    else:
        print(f"请求第{page}页失败，状态码: {response.status_code}")

# 第三步：保存评论到CSV文件
if all_comments:
    csv_file = 'music_comments.csv'
    with open(csv_file, 'w', newline='', encoding='utf-8-sig') as f:
        fieldnames = ['用户名', '评论时间', '评论内容']
        f_csv = csv.DictWriter(f, fieldnames=fieldnames)
        f_csv.writeheader()
        f_csv.writerows(all_comments)
    
    print(f"\n爬取完成！")
    print(f"共收集到 {len(all_comments)} 条评论")
    print(f"数据已保存到 {csv_file}")
else:
    print("未收集到任何评论数据")
```
